\begin{rSection}{Research Experience}
\begin{rSubsection}{\href{https://fluentrobotics.com}{Fluent Robotics Lab, University of Michigan}}{Ann Arbor, USA}{Graduate Research Assistant; Advisor: Dr. Christoforos Mavrogiannis}{Aug 2024 -- Present}
\item {\bf Legible robot navigation:} Developed MPPI-based intent-expression algorithms in PyTorch and deployed them on Stretch 2. Evaluated intent representation and pedestrian distraction in two controlled studies ($N=45$ each). First-author paper accepted to {\bf THRI 2026}. [\href{https://github.com/fluentrobotics/Legible_MPPI}{Code}]
\item {\bf Prediction for decision-making:} Evaluated how human motion prediction quality affects navigation across planners and predictors in constrained spaces, linking prediction metrics to robot performance and human experience ({\bf HRI 2026}).
\item {\bf Integrated robot systems:} Developing Alpaca, an interactive social-navigation platform integrating pedestrian tracking, prediction, navigation, and conversational interaction in ROS 2. First-author manuscript {\bf under review at HRI 2027}.
\item {\bf Data for social navigation:} Contributing to Bi3, spanning two robot platforms and two cultural contexts ({\bf ICRA 2026}), and ACME, a multi-cultural, multi-embodiment dataset ({\bf under review at~IJRR}).
\end{rSubsection}

\begin{rSubsection}{\href{https://clear-nus.github.io}{CLeAR Lab, National University of Singapore}}{Singapore}{Research Assistant; Advisor: \href{https://haroldsoh.com}{Dr. Harold Soh}}{Aug 2023 -- Aug 2024}
\item {\bf Diffusion planning on hardware:} Deployed LTLDoG, a diffusion-based planner with temporal-logic constraints, as a global planner for Unitree Go2 navigation experiments ({\bf RA-L 2024}).
\item {\bf Scenario-based evaluation:} Developed an LLM-driven ROS 2/Gazebo tool to generate social navigation scenarios for benchmarking planners and identifying failure cases ({\bf SocRATES, RA-L 2026}; runner-up best paper, RSS 2024 workshop). [\href{https://github.com/clear-nus/SocRATES}{Code}]
\item Integrated and deployed quadrupedal navigation systems for experiments and data collection in dense, unstructured human environments, including university canteens.
\end{rSubsection}
